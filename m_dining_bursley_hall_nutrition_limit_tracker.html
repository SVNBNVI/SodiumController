Active Cron-Job & Daily Menu Sync Engine

In-Browser Active Scheduler + CORS Scraper

Polls U-M Bursley Dining endpoint for the active date, parses stations & nutrition facts, caches in localStorage, and falls back to the verified Bursley rotation dataset if CORS-blocked.

Live Cron Execution Logs Cache: Warmed
Production Scheduled Cron Worker (0 4 * * * Daily at 4:00 AM EST) 
// .github/workflows/bursley-menu-cron.yml & worker/syncBursleyMenu.js
import cron from 'node-cron';
import * as cheerio from 'cheerio';
import fs from 'fs/promises';

// Runs every morning at 04:00 AM Ann Arbor time (America/Detroit)
cron.schedule('0 4 * * *', async () => {
  const today = new Date().toISOString().split('T')[0];
  const url = `https://dining.umich.edu/menus-locations/dining-halls/bursley/?menuDate=${today}`;
  console.log(`[CRON] Fetching Bursley Hall Menu for ${today}...`);

  const res = await fetch(url, { headers: { 'User-Agent': 'UMich-Bursley-Nutrition-Tracker/1.0' } });
  const html = await res.text();
  const $ = cheerio.load(html);
  const items = [];

  $('.courses_wrapper .course').each((_, stationEl) => {
    const station = $(stationEl).find('h4').text().trim();
    $(stationEl).find('.item-name').each((__, itemEl) => {
      // Extracts FDA nutrition table values (Calories, Sodium, Carbs, Sat Fat, Added Sugar, Protein)
      items.push({ station, name: $(itemEl).text().trim(), date: today });
    });
  });

  await fs.writeFile(`./public/data/bursley-${today}.json`, JSON.stringify({ updatedAt: new Date(), items }, null, 2));
  console.log(`[CRON] Cached ${items.length} Bursley dishes for ${today}.`);
}, { timezone: 'America/Detroit' });
